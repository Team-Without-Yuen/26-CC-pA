# 編譯指令(windows)(UCRT64) 
g++ -std=c++11 -I. -Ilib main.cpp src/core/*.cpp src/io/*.cpp src/analysis/*.cpp -L./include/lib/win -lcadical -o NetlistTool

# 編譯指令(linux) 
g++ -std=c++11 -I. -Ilib main.cpp src/core/*.cpp src/io/*.cpp src/analysis/*.cpp -L./include/lib/linux -lcadical -o NetlistTool

# 在 Windows下執行指令：
./NetlistTool.exe test01.v

# 在 Linux 下執行指令：
./NetlistTool.exe test01.v

# (輸出與輸入一樣就能做到修改)
./NetlistTool.exe test01.v output.v
